## 三、状态管理方案 Pinia

### 3.1 Store 设计原则

- 按业务领域划分 Store，职责单一
- API 调用放在 `api/` 层，Store 只做编排
- 持久化数据存 localStorage；临时状态（SSE 连接、UI 态）不持久化

### 3.2 核心 Store 定义

| Store | State | Actions | 持久化 |
|-------|-------|---------|--------|
| `user` | token, refreshToken, userInfo, loginLoading | login(), fetchUserInfo(), logout() | token 存 localStorage |
| `project` | projectList, currentProject, loading | fetchList(), createProject(), getDetail() | 否 |
| `workflow` | stages[], sseConnection, isConnected | connectSse(), disconnectSse(), fetchLogs() | 否 |
| `app` | sidebarCollapsed, theme | toggleSidebar() | sidebarCollapsed 存 localStorage |

### 3.3 用户 Store 设计

```typescript
// stores/modules/user.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UserInfo, LoginRequest } from '@/types'

export const useUserStore = defineStore('user', () => {
  const token = ref(localStorage.getItem('token') || '')
  const userInfo = ref<UserInfo | null>(null)
  const loginLoading = ref(false)
  const isLoggedIn = computed(() => !!token.value)

  async function login(params: LoginRequest) {
    loginLoading.value = true
    try {
      const res = await authApi.login(params)
      token.value = res.accessToken
      userInfo.value = res.user
      localStorage.setItem('token', res.accessToken)
    } finally {
      loginLoading.value = false
    }
  }

  function logout() {
    token.value = ''
    userInfo.value = null
    localStorage.removeItem('token')
  }

  return { token, userInfo, loginLoading, isLoggedIn, login, logout }
})
```

### 3.4 工作流 Store（含 SSE 管理）

```typescript
// stores/modules/workflow.ts
export const useWorkflowStore = defineStore('workflow', () => {
  const stages = ref<WorkflowStage[]>([])
  const sseSource = ref<EventSource | null>(null)
  const isConnected = ref(false)

  const progress = computed(() => {
    const done = stages.value.filter(s => s.status === 'COMPLETED').length
    return stages.value.length ? Math.round((done / stages.value.length) * 100) : 0
  })

  function connectSse(projectId: string) {
    disconnectSse()
    const es = new EventSource(`/api/v1/workflow/${projectId}/stream`)
    es.addEventListener('stage:start', (e) => updateStage(JSON.parse(e.data), 'RUNNING'))
    es.addEventListener('stage:complete', (e) => updateStage(JSON.parse(e.data), 'COMPLETED'))
    es.addEventListener('stage:error', (e) => updateStage(JSON.parse(e.data), 'FAILED'))
    es.addEventListener('clarification:needed', () => { /* 由 ClarifyPanel 组件监听处理 */ })
    es.onerror = () => { isConnected.value = false }
    sseSource.value = es
    isConnected.value = true
  }

  function disconnectSse() {
    sseSource.value?.close()
    sseSource.value = null
    isConnected.value = false
  }

  return { stages, isConnected, progress, connectSse, disconnectSse }
})
```

### 4.2 路由守卫

```typescript
// router/guards.ts
import type { Router } from 'vue-router'
import { useUserStore } from '@/stores'

export function setupGuards(router: Router) {
  router.beforeEach((to, from, next) => {
    // 1. 设置页面标题
    document.title = to.meta.title ? `${to.meta.title} - AI PickUp` : 'AI PickUp'

    // 2. 认证检查
    const userStore = useUserStore()
    if (to.meta.requiresAuth && !userStore.isLoggedIn) {
      next({ name: 'Login', query: { redirect: to.fullPath } })
      return
    }
    if (to.name === 'Login' && userStore.isLoggedIn) {
      next({ name: 'Dashboard' })
      return
    }
    next()
  })
}
```

## 六、接口请求层封装 Axios

### 6.1 Axios 实例 + 拦截器

```typescript
// api/index.ts
import axios, { type AxiosInstance, type InternalAxiosRequestConfig } from 'axios'
import { useUserStore } from '@/stores'
import { ElMessage } from 'element-plus'
import router from '@/router'
import type { ApiResponse } from '@/types'

const http: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE + '/api/v1',
  timeout: 30000,
  headers: { 'Content-Type': 'application/json' }
})

// 请求拦截器：注入 Token + TraceId
http.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const userStore = useUserStore()
  if (userStore.token) {
    config.headers.Authorization = `Bearer ${userStore.token}`
  }
  config.headers['X-Request-Id'] = crypto.randomUUID()
  return config
})

// 响应拦截器：统一错误处理
http.interceptors.response.use(
  (response) => {
    const { code, message, data } = response.data as ApiResponse
    if (code === 200) return data as any
    if (code === 401) {
      useUserStore().logout()
      router.push({ name: 'Login' })
    }
    ElMessage.error(message || '请求失败')
    return Promise.reject(new Error(message))
  },
  (error) => {
    if (error.response?.status === 401) {
      useUserStore().logout()
      router.push({ name: 'Login' })
    } else if (error.response?.status === 429) {
      ElMessage.error('请求过于频繁，请稍后重试')
    } else {
      ElMessage.error(error.message || '网络错误')
    }
    return Promise.reject(error)
  }
)

export default http
```

## 七、SSE 实时推送封装

### 7.1 useSSE Composable

```typescript
// composables/useSSE.ts
import { ref, onUnmounted } from 'vue'

interface SseOptions {
  onStageStart?: (data: any) => void
  onStageProgress?: (data: any) => void
  onStageComplete?: (data: any) => void
  onStageError?: (data: any) => void
  onClarificationNeeded?: (data: any) => void
  onWorkflowComplete?: (data: any) => void
  onError?: (error: Event) => void
}

export function useSSE() {
  const isConnected = ref(false)
  let eventSource: EventSource | null = null

  function connect(projectId: string, options: SseOptions) {
    disconnect()
    const url = `/api/v1/workflow/${projectId}/stream`
    eventSource = new EventSource(url)
    isConnected.value = true

    eventSource.addEventListener('stage:start', (e) => options.onStageStart?.(JSON.parse(e.data)))
    eventSource.addEventListener('stage:progress', (e) => options.onStageProgress?.(JSON.parse(e.data)))
    eventSource.addEventListener('stage:complete', (e) => options.onStageComplete?.(JSON.parse(e.data)))
    eventSource.addEventListener('stage:error', (e) => options.onStageError?.(JSON.parse(e.data)))
    eventSource.addEventListener('clarification:needed', (e) => options.onClarificationNeeded?.(JSON.parse(e.data)))
    eventSource.addEventListener('workflow:complete', (e) => options.onWorkflowComplete?.(JSON.parse(e.data)))

    eventSource.onerror = (e) => {
      isConnected.value = false
      options.onError?.(e)
    }
  }

  function disconnect() {
    eventSource?.close()
    eventSource = null
    isConnected.value = false
  }

  onUnmounted(() => disconnect())

  return { isConnected, connect, disconnect }
}
```

### 7.2 页面中使用

```vue

  connect(route.params.id as string, {
    onStageStart: (d) => workflowStore.updateStage(d.stage, 'RUNNING'),
    onStageComplete: (d) => workflowStore.updateStage(d.stage, 'COMPLETED'),
    onStageError: (d) => workflowStore.updateStage(d.stage, 'FAILED', d.error),
    onClarificationNeeded: (d) => { questions.value = d.questions },
    onWorkflowComplete: () => { ElMessage.success('项目生成完成！') },
  })
})

onUnmounted(() => disconnect())
</script>
```

### 7.3 SSE 事件类型

| 事件名 | 触发时机 | Payload |
|--------|----------|---------|
| `stage:start` | 工作流节点开始执行 | `{ stage: string, order: number }` |
| `stage:progress` | 节点执行中推送进度 | `{ stage: string, progress: number, message: string }` |
| `stage:complete` | 节点执行完成 | `{ stage: string, output: object }` |
| `stage:error` | 节点执行失败 | `{ stage: string, error: string, retryable: boolean }` |
| `clarification:needed` | 需要用户澄清需求 | `{ questions: ClarifyQuestion[] }` |
| `workflow:complete` | 整个工作流完成 | `{ projectId: string }` |

## 九、TypeScript 类型体系

### 9.1 类型文件划分

```typescript
// types/api.d.ts — 通用 API 响应
export interface ApiResponse<T = any> {
  code: number
  message: string
  data: T
  timestamp: string
  traceId: string
}

export interface PageResult<T> {
  records: T[]
  total: number
  page: number
  size: number
  totalPages: number
}

// types/project.d.ts — 项目领域类型
export interface CreateProjectRequest {
  name: string
  requirementText: string
  projectType?: 'WEB_APP' | 'MINI_PROGRAM' | 'MOBILE_APP' | 'API_SERVICE'
  techPreferences?: {
    frontend?: string
    backend?: string
    database?: string
  }
  autoConfirm?: boolean
}

export type ProjectStatus =
  | 'DRAFT' | 'REQUIREMENT_PARSING' | 'REQUIREMENT_CONFIRMATION'
  | 'DOC_GENERATING' | 'PROTOTYPE_GENERATING' | 'UI_DESIGNING'
  | 'TECH_PLAN_GENERATING' | 'PLAN_INTEGRATING'
  | 'CODE_GENERATING' | 'QUALITY_CHECKING'
  | 'COMPLETED' | 'FAILED'

export interface ProjectResponse {
  projectId: string
  name: string
  status: ProjectStatus
  progress: number
  currentStage: string
  createdAt: string
}
```

### 9.2 类型使用原则

- **接口 `interface`**：用于对象结构定义（Props、DTO、响应体）
- **类型别名 `type`**：用于联合类型、映射类型
- **枚举 `enum`**：仅用于有固定值集合的常量（避免运行时开销的场景用 `const` 对象）
- **泛型**：通用组件（如 `AppTable<T>`）使用泛型约束
