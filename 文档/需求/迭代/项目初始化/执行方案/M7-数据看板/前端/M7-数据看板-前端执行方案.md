## 3. StatCard.vue

```typescript
interface StatCardProps {
  title: string        // 卡片标题
  value: number        // 统计数值
  icon: string         // Element Plus 图标名
  trend?: number       // 趋势百分比（正=升，负=降）
```

## 4. TrendChart.vue（基于 ECharts）

```typescript
interface TrendChartProps {
  title: string          // 图表标题
  data: DailyCount[]     // [{ date: "2026-05-12", count: 8 }, ...]
  xKey?: string          // 默认 "date"
  yKey?: string          // 默认 "count"
  color?: string         // 折线颜色
}
```

**ECharts 配置**：
- type: 'line' + smooth 平滑曲线
- areaStyle 半透明填充
- tooltip 十字准星
- xAxis: 日期格式化 'MM-DD'
- dataZoom: 支持缩放

## 7. Pinia Store - dashboardStore.ts

```typescript
export const useDashboardStore = defineStore('dashboard', () => {
  const overview = ref<OverviewStats | null>(null)
  const projectTrend = ref<DailyCount[]>([])
  const stageStats = ref<StageStat[]>([])
  const templateUsage = ref<TemplateUsage[]>([])
  const userRanking = ref<UserRank[]>([])
  const aiCallStats = ref<AiCallStat[]>([])

  async function fetchOverview() {
    const res = await api.get('/admin/stats/overview')
    overview.value = res.data
  }

  async function fetchProjectTrend(days = 30) {
    const res = await api.get('/admin/stats/project-trend', { params: { days } })
    projectTrend.value = res.data
  }

  async function fetchAll() {
    await Promise.all([
      fetchOverview(),
      fetchProjectTrend(),
      fetchStageStats(),
      fetchTemplateUsage(),
      fetchUserRanking(),
      fetchAiCallStats()
    ])
  }

  return { overview, projectTrend, stageStats, templateUsage,
    userRanking, aiCallStats, fetchOverview, fetchProjectTrend, fetchAll }
})
```
