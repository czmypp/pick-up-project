## 8. SSE 流式连接 - useSseStream.ts

```typescript
export function useSseStream(projectId: string) {
  let eventSource: EventSource | null = null

  function connect() {
    eventSource = new EventSource(`/api/v1/workflow/${projectId}/stream`)

    eventSource.addEventListener('stage:start', (e) => {
      const data = JSON.parse(e.data)
      ElNotification.info(`开始生成: ${data.stage}`)
    })

    eventSource.addEventListener('stage:progress', (e) => {
      // 更新进度条
    })

    eventSource.addEventListener('stage:complete', (e) => {
      ElNotification.success('生成完毕')
    })

    eventSource.addEventListener('preview:ready', (e) => {
      // 展示「预览就绪」通知，加载 changeLogs + subtasks
    })

    eventSource.addEventListener('update:start', (e) => {
      // previewStatus = UPDATING，显示 loading 遮罩
    })

    eventSource.addEventListener('update:complete', (e) => {
      // previewStatus = PREVIEWING，刷新版本 + 变更记录
    })

    eventSource.addEventListener('phase:confirmed', (e) => {
      // 阶段已确认，自动跳转下一阶段
    })

    eventSource.addEventListener('stage:error', (e) => {
      ElMessage.error('生成失败，请重试')
    })

    eventSource.addEventListener('workflow:complete', (e) => {
      // 工作流完成，跳转项目详情
    })

    eventSource.onerror = () => {
      // 自动重连（延迟 3 秒）
      setTimeout(connect, 3000)
    }
  }

  function disconnect() {
    eventSource?.close()
  }

  return { connect, disconnect }
}
```

## 10. SSE 事件流处理完整映射

| 事件类型 | 前端动作 |
|----------|----------|
| `stage:start` | 更新阶段状态为 RUNNING |
| `stage:progress` | 更新子任务进度条 |
| `stage:complete` | 更新阶段状态为 COMPLETED |
| `preview:ready` | 展示「预览就绪」通知，加载 changeLogs + subtasks |
| `update:start` | previewStatus = UPDATING，显示 loading 遮罩 |
| `update:complete` | previewStatus = PREVIEWING，刷新版本 + 变更记录 |
| `phase:confirmed` | 阶段已确认，自动跳转下一阶段 |
| `stage:error` | ElMessage.error(errorMsg)，显示重试按钮 |
| `workflow:complete` | 工作流完成，跳转项目详情 |
