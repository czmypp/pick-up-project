## 3. 三步创建向导数据流

```
Step1TemplateSelect.vue
    │ mounted → GET /api/v1/templates
    │           → 展示模板卡片列表
    │ 用户选择 → emit('select', templateId)
    │           → wizardContext.templateId = templateId
    │
    ▼
Step2ModuleCustomize.vue
    │ mounted → GET /api/v1/templates/{templateId}/modules
    │           → 展示模块列表（必选/可选）
    │ 用户勾选 → wizardContext.selectedModuleIds = [...]
    │
    ▼
Step3LayoutConfig.vue
    │ mounted → GET /api/v1/templates/{templateId}/layouts
    │           → 展示排版方案列表
    │ 用户选择 → wizardContext.layoutId = layoutId
    │ 用户点击「创建」→ POST /api/v1/projects (wizardContext)
    │           → 成功 → router.push(`/projects/${id}`)
    │           → 失败 → 错误提示（如配额不足）
```

## 7. Pinia Store - projectStore.ts

```typescript
export const useProjectStore = defineStore('project', () => {
  // State
  const projectList = ref<Project[]>([])
  const currentProject = ref<ProjectDetail | null>(null)
  const createWizard = ref({
    step: 1,
    templateId: '',
    selectedModules: [] as string[],
    layoutId: '',
    layoutConfigOverride: {}
  })

  // Actions
  async function fetchProjects(params: ProjectQuery) {
    const res = await api.get('/projects', { params })
    projectList.value = res.data.records
    return res.data
  }

  async function fetchProjectDetail(id: string) {
    const res = await api.get(`/projects/${id}`)
    currentProject.value = res.data
    return res.data
  }

  async function createProject(data: CreateProjectRequest) {
    const res = await api.post('/projects', data)
    return res.data
  }

  async function updateProject(id: string, data: Partial<CreateProjectRequest>) {
    const res = await api.put(`/projects/${id}`, data)
    return res.data
  }

  async function deleteProject(id: string) {
    await api.delete(`/projects/${id}`)
    projectList.value = projectList.value.filter(p => p.id !== id)
  }

  return { projectList, currentProject, createWizard,
    fetchProjects, fetchProjectDetail, createProject, updateProject, deleteProject }
})
```
